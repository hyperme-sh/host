cmake_minimum_required(VERSION 3.22)
project(hypermesh_host_crash LANGUAGES C CXX)
find_package(sentry CONFIG REQUIRED)
if(NOT HOST_RUST_ARCHIVE)
  message(FATAL_ERROR "HOST_RUST_ARCHIVE is required")
endif()
add_library(hypermesh_host_crash SHARED entry.c)
set_target_properties(hypermesh_host_crash PROPERTIES LINKER_LANGUAGE CXX INSTALL_RPATH "$ORIGIN/../lib")
target_link_libraries(hypermesh_host_crash PRIVATE ${HOST_RUST_ARCHIVE} sentry::sentry ${CMAKE_DL_LIBS} pthread)
if(APPLE)
  target_link_libraries(hypermesh_host_crash PRIVATE "-framework Security" "-framework CoreFoundation" iconv)
endif()
add_executable(host-crash-options-contract options-test.c)
set_target_properties(host-crash-options-contract PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(host-crash-options-contract PRIVATE sentry::sentry ${CMAKE_DL_LIBS})
install(TARGETS hypermesh_host_crash LIBRARY DESTINATION bin)
